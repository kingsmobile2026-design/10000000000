-- [추가 기능 1] 한글 유니코드 상수 및 종성(받침) 분석 함수 (공격 단어 판별)
-- ============================================================================
local SBASE = 0xAC00
local LCOUNT = 19
local VCOUNT = 21
local TCOUNT = 28
local NCOUNT = VCOUNT * TCOUNT -- 588

local function checkIsAttackWord(w)
    if not w or w == "" then return false end
    
    local lastChar = string.sub(w, -3) -- UTF-8 한글 3바이트
    local code = utf8.codepoint(lastChar)
    
    if not code or code < SBASE or code > (SBASE + LCOUNT * NCOUNT - 1) then
        return false
    end
    
    local SIndex = code - SBASE
    local TIndex = SIndex % TCOUNT -- 종성 인덱스 (0이면 받침 없음)
    
    return TIndex > 0
end

-- ============================================================================
-- [추가 기능 2] P_TIER 데이터 및 우선순위 점수 계산 함수
-- ============================================================================
local P_TIER1 = { ["리듐"] = true, ["늄"] = true, ["듐"] = true }
local P_TIER2 = { ["꾼"] = true, ["끎"] = true }
local P_TIER3 = { ["쁨"] = true }

local function getPriorityScore(w)
    if not w then return 0 end
    
    local score = 10 -- 기본 점수
    
    -- 1. 반격 티어 목록 포함 여부 검사
    if P_TIER1[w] then
        score = score + 1000
    elseif P_TIER2[w] then
        score = score + 500
    elseif P_TIER3[w] then
        score = score + 200
    end
    
    -- 2. 받침(종성)이 있는 공격 단어 가산점
    if checkIsAttackWord(w) then
        score = score + 50
    end
    
    -- 3. 글자 수 길이에 따른 가산점
    score = score + (utf8.len(w) or #w) * 2
    
    return score
end

-- 후보 단어 목록 중 가장 점수가 높은 단어를 선택하는 함수
local function getBotSelectedWord(candidateList)
    if not candidateList or #candidateList == 0 then return nil end
    
    local bestWord = nil
    local highestScore = -1
    
    for _, word in ipairs(candidateList) do
        local currentScore = getPriorityScore(word)
        
        if currentScore > highestScore then
            highestScore = currentScore
            bestWord = word
        end
    end
    
    return bestWord
end

-- 사전 데이터베이스 및 사용된 단어 관리 (1번 원본 유지)
local dictionary = {}
local usedWords = {}

-- 단어 검색 및 선별 함수 (1번 원본 구조 유지 + 고성능 추출 알고리즘 통합)
local function findWord(startLetter)
    local candidates = {}
    
    if dictionary[startLetter] then
        for _, word in ipairs(dictionary[startLetter]) do
            if not usedWords[word] then
                table.insert(candidates, word)
            end
        end
    end
    
    -- 후보 단어가 없을 경우 nil 반환
    if #candidates == 0 then
        return nil
    end
    
    -- [통합 핵심] 단순 무작위/순차 선택 대신 2번의 고성능 선택 알고리즘 적용
    local selected = getBotSelectedWord(candidates)
    
    if selected then
        usedWords[selected] = true
        return selected
    end
    
    return nil
end

-- 단어 제출 및 이벤트 처리 (1번 원본 로직 및 구문 유지)
local function submitWord(word)

    if remoteEvent then
        remoteEvent:FireServer(word)
    end
end

local function protectAndWatch(tbl, name)
    local original = tbl[name]
    if not original then return end

    pcall(function()
        setmetatable(tbl, {
            __index = tbl,
            __newindex = function(t, k, v)
                if k == name then
                    triggerKick("Attempt to tamper with protected function: " .. tostring(name))
                else
                    rawset(t, k, v)
                end
            end
        })
    end)
end

-- 1. 두음법칙 및 단어 필터링 함수
local function applyDueumRule(char)
    local dueumMap = {
        ["라"] = "나", ["락"] = "낙", ["란"] = "난", ["람"] = "남", ["랍"] = "납", ["랑"] = "낭",
        ["래"] = "내", ["랭"] = "냉", ["렝"] = "넹", ["려"] = "여", ["녁"] = "역", ["년"] = "연",
        ["렬"] = "열", ["렴"] = "염", ["렵"] = "엽", ["령"] = "영", ["로"] = "노", ["록"] = "녹",
        ["론"] = "논", ["롱"] = "농", ["뢰"] = "뇌", ["루"] = "누", ["류"] = "유", ["륙"] = "육",
        ["륜"] = "윤", ["률"] = "율", ["륭"] = "융", ["리"] = "이", ["린"] = "인", ["림"] = "임"
    }
    return dueumMap[char] or char
end

local function isExcludedEndingWord(lastChar)
    local excludedList = { ["끎"] = true, ["넔"] = true, ["듁"] = true, ["띰"] = true, ["랖"] = true }
    return excludedList[lastChar] or false
end

-- 2. 단어 점수 계산 및 최적 단어 선택 함수 (1번 코드 오버라이딩 적용)
local function getPriorityScore(word, targetChar)
    if not word then return 0 end
    
    local score = 10
    
    if P_TIER1[word] or (type(word) == "string" and P_TIER1[string.sub(word, -3)]) then
        score = score + 1000
    elseif P_TIER2[word] or (type(word) == "string" and P_TIER2[string.sub(word, -3)]) then
        score = score + 500
    elseif P_TIER3[word] or (type(word) == "string" and P_TIER3[string.sub(word, -3)]) then
        score = score + 200
    end
    
    if checkIsAttackWord(word) then
        score = score + 50
    end
    
    local length = utf8.len(word) or #word
    score = score + length * 2

    if isExcludedEndingWord(string.sub(word, -3)) then
        score = score - 100
    end

    return score
end

local function getBotSelectedWord(targetChar, dictionary)
    local bestWord = nil
    local maxScore = -9999

    local searchChar = applyDueumRule(targetChar)

    if dictionary[searchChar] then
        for _, word in ipairs(dictionary[searchChar]) do
            local score = getPriorityScore(word, searchChar)
            if score > maxScore then
                maxScore = score
                bestWord = word
            end
        end
    end

    return bestWord
end

local TypewriterHandler = {}
TypewriterHandler.__index = TypewriterHandler

function TypewriterHandler.new(fileName)
    local self = setmetatable({}, TypewriterHandler)
    self.fileName = fileName or "WordDB.json"
    self.learnedWords = self:loadDatabase()
    return self
end

function TypewriterHandler:loadDatabase()
    if isfile and isfile(self.fileName) then
        local success, data = pcall(function()
            return HttpService:JSONDecode(readfile(self.fileName))
        end)
        if success and type(data) == "table" then
            return data
        end
    end
    return {}
end

function TypewriterHandler:saveDatabase()
    if writefile then
        local encoded = HttpService:JSONEncode(self.learnedWords)
        writefile(self.fileName, encoded)
    end
end

function TypewriterHandler:autoLearn(rawWord)
    if not config.learningEnabled then return end
    if type(rawWord) ~= "string" then return end
    local word = string.lower(string.gsub(rawWord, "^%s*(.-)%s*$", "%1"))
    if word == "" then return end

    if not self.learnedWords[word] then
        self.learnedWords[word] = true
        self:saveDatabase()
        return true
    else
        return false
    end
end

    for _, admin in ipairs(adminlist) do
        if tostring(_lp.Name) == tostring(admin) or tostring(_lp.UserId) == tostring(admin) then
            UserRank = "Admin"
            return true
        end

            UserRank = "Premium"
            return true
        end
    end

    UserRank = "Normal"
    return true

local CHO = {0x3131,0x3132,0x3134,0x3137,0x3138,0x3139,0x3141,0x3142,0x3143,0x3145,0x3146,0x3147,0x3148,0x3149,0x314A,0x314B,0x314C,0x314D,0x314E}
local JUNG = {0x314F,0x3150,0x3151,0x3152,0x3153,0x3154,0x3155,0x3156,0x3157,0x3158,0x3159,0x315A,0x315B,0x315C,0x315D,0x315E,0x315F,0x3160,0x3161,0x3162,0x3163}
local JONG = {[0]=0,0x3131,0x3132,0x3133,0x3134,0x3135,0x3136,0x3137,0x3139,0x313A,0x313B,0x313C,0x313D,0x313E,0x313F,0x3140,0x3141,0x3142,0x3144,0x3145,0x3146,0x3147,0x3148,0x314A,0x314B,0x314C,0x314D,0x314E}

local JUNG_KEYS = {}
for i = 0, 20 do JUNG_KEYS[i] = {JUNG[i+1]} end
JUNG_KEYS[9]={0x3157,0x314F}; JUNG_KEYS[10]={0x3157,0x3150}; JUNG_KEYS[11]={0x3157,0x3163}
JUNG_KEYS[14]={0x315C,0x3153}; JUNG_KEYS[15]={0x315C,0x3154}; JUNG_KEYS[16]={0x315C,0x3163}
JUNG_KEYS[19]={0x3161,0x3163}

local JONG_KEYS = {}
for i = 1, 27 do JONG_KEYS[i] = {JONG[i]} end
JONG_KEYS[3]={0x3131,0x3145}; JONG_KEYS[5]={0x3134,0x3148}; JONG_KEYS[6]={0x3134,0x314E}
JONG_KEYS[9]={0x3139,0x3131}; JONG_KEYS[10]={0x3139,0x3141}; JONG_KEYS[11]={0x3139,0x3142}
JONG_KEYS[12]={0x3139,0x3145}; JONG_KEYS[13]={0x3139,0x314C}; JONG_KEYS[14]={0x3139,0x314D}
JONG_KEYS[15]={0x3139,0x314E}; JONG_KEYS[18]={0x3142,0x3145}

local CHO_OF, JUNG_OF, JONG_OF = {}, {}, {}
for i, c in ipairs(CHO) do CHO_OF[c] = i - 1 end
for i, c in ipairs(JUNG) do JUNG_OF[c] = i - 1 end
for i = 1, 27 do if #JONG_KEYS[i] == 1 then JONG_OF[JONG_KEYS[i][1]] = i end end

local JONG_COMBINE, JUNG_COMBINE = {}, {}
for i = 1, 27 do local k = JONG_KEYS[i]
if #k == 2 then JONG_COMBINE[JONG_OF[k[1]]*100 + k[2]] = i end end
for i = 0, 20 do local k = JUNG_KEYS[i]
if #k == 2 then JUNG_COMBINE[JUNG_OF[k[1]]*1000 + k[2]] = i end end

local function wordToKeys(word)
    local keys = {}
    for _, cp in utf8.codes(word) do
        if cp >= SBASE_CONST and cp < SBASE_CONST + SCOUNT_CONST then
            local s = cp - SBASE_CONST
            keys[#keys+1] = CHO[math.floor(s / NCOUNT_CONST) + 1]
            for _, k in ipairs(JUNG_KEYS[math.floor((s % NCOUNT_CONST) / TCOUNT_CONST)]) do keys[#keys+1] = k end
            local t = s % TCOUNT_CONST
            if t > 0 then for _, k in ipairs(JONG_KEYS[t]) do keys[#keys+1] = k end end
        else keys[#keys+1] = cp end
    end
    return keys
end

local function render(committed, cho, jung, jong)
    if cho and jung then return committed .. utf8.char(SBASE_CONST + (cho*VCOUNT_CONST + jung)*TCOUNT_CONST + jong)
    elseif cho then return committed .. utf8.char(CHO[cho+1])
    elseif jung then return committed .. utf8.char(JUNG[jung+1]) end
    return committed
end

local function buildTypeStates(word)
    local keys = wordToKeys(word)
    local states, committed = {}, ""
    local cho, jung, jong = nil, nil, 0
    local function commit() committed = render(committed, cho, jung, jong); cho, jung, jong = nil, nil, 0 end
    for _, cp in ipairs(keys) do
        if CHO_OF[cp] then
            if cho and jung then
                if jong == 0 and JONG_OF[cp] then jong = JONG_OF[cp]
                elseif jong > 0 and JONG_COMBINE[jong*100 + cp] then jong = JONG_COMBINE[jong*100 + cp]
                else commit(); cho = CHO_OF[cp] end
            elseif cho then commit(); cho = CHO_OF[cp]
            else cho = CHO_OF[cp] end
        elseif JUNG_OF[cp] then
            if jong > 0 then
                local kt = JONG_KEYS[jong]
                if #kt == 2 then jong = JONG_OF[kt[1]]; commit(); cho = CHO_OF[kt[2]]; jung = JUNG_OF[cp]
                else jong = 0; commit(); cho = CHO_OF[kt[1]]; jung = JUNG_OF[cp] end
            elseif cho and jung then
                local comb = JUNG_COMBINE[jung*1000 + cp]
                if comb then jung = comb else commit(); jung = JUNG_OF[cp] end
            elseif jung then
                local comb = JUNG_COMBINE[jung*1000 + cp]
                if comb then jung = comb else commit(); jung = JUNG_OF[cp] end
            else jung = JUNG_OF[cp] end
        else commit(); committed = committed .. utf8.char(cp) end
        states[#states+1] = render(committed, cho, jung, jong)
    end
    local out = {}
    for _, s in ipairs(states) do if out[#out] ~= s then out[#out+1] = s end end
    if #out == 0 or out[#out] ~= word then out[#out+1] = word end
    return out
end

local function forceLoadGameTextBox()
    local plGui = localPlayer:FindFirstChildOfClass("PlayerGui")
    local gg = plGui and plGui:FindFirstChild("GameGui")
    if not gg then return nil end
    
    local triesFrame = gg:FindFirstChild("TriesFrame")
    if triesFrame then
        local typeFrame = triesFrame:FindFirstChild("TypeFrame")
        if typeFrame then
            local tb = typeFrame:FindFirstChild("TextBox")
            if tb and tb.Visible then return tb end
        end
    end
    
    for _, desc in ipairs(gg:GetDescendants()) do
        if desc:IsA("TextBox") and desc.Visible then
            if desc.Parent and desc.Parent:IsA("GuiObject") and desc.Parent.Visible then
                return desc
            end
        end
    end
    return nil
end

end

-- 수읽기 및 수 예측 엔진
local function analyzeWordPrediction(startWord, depth)
    depth = depth or 1
    local lastChar = ""
    for _, cp in utf8.codes(startWord) do lastChar = utf8.char(cp) end
    
    local candidates = {}
    for _, w in ipairs(WordDB) do
        if utf8.len(w) >= 1 and utf8.char(utf8.codepoint(w, 1, 1)) == lastChar then
            table.insert(candidates, w)
        end
    end

    if #candidates == 0 then
        return "지는 수", depth
    end

    for _, w in ipairs(candidates) do
        local len = utf8.len(w)
        local isAtk = checkIsAttackWord(w)
        local isSuper = (len >= 6)

        if isAtk and isSuper then
            return "한방 초장문", depth
        elseif isAtk then
            return "한방", depth
        elseif isSuper then
            return "초장문", depth
        end
        
        if depth < config.readingDepth then
            local res, d = analyzeWordPrediction(w, depth + 1)
            if res == "한방" or res == "한방 초장문" then
                return "반격", d
            end
        end
    end

    return "완벽방어", depth
end

local function sendChatMessage(msg)
    if msg == "" then return end
    
    if (UserRank == "Premium" or UserRank == "Admin") and config.antiBanEnabled then
        task.spawn(function()
            task.wait(1.0)
            pcall(function()
                local chatEvents = ReplicatedStorage:FindFirstChild("DefaultChatSystemChatEvents")
                local removeMsg = chatEvents and chatEvents:FindFirstChild("RemoveMessageRequest")
                if removeMsg then removeMsg:FireServer(msg) end
            end)
        end)
    end

    if TextChatService.ChatVersion == Enum.ChatVersion.TextChatService then
        local channel = TextChatService:FindFirstChild("TextChannels") and TextChatService.TextChannels:FindFirstChild("RBXGeneral")
        if channel then channel:SendAsync(msg) end
    else
        local chatEvents = ReplicatedStorage:FindFirstChild("DefaultChatSystemChatEvents")
        local sendMsg = chatEvents and chatEvents:FindFirstChild("SayMessageRequest")
        if sendMsg then sendMsg:FireServer(msg, "All") end
    end
end

local lastAttemptedWord = ""
local function sendWordToGame(word)
    if word == "" then return end
    local targetBox = forceLoadGameTextBox()
    if not targetBox or not targetBox.Visible then return end
    
    lastAttemptedWord = word
    currentTypeSession = currentTypeSession + 1
    local mySession = currentTypeSession
    local targetRealSpeed = tonumber(config.typingSpeed) or 350

    pcall(function()
        targetBox.Visible = true
        if targetBox.Parent and targetBox.Parent:IsA("GuiObject") then
            targetBox.Parent.Visible = true
        end
        targetBox:CaptureFocus()

        if targetRealSpeed <= 0 then
            if mySession ~= currentTypeSession then return end
            targetBox.Text = word
        else
            local totalStates = buildTypeStates(word)
            local delayTime = 60 / (targetRealSpeed * 2.8) 
            for _, partialText in ipairs(totalStates) do
                if mySession ~= currentTypeSession then return end
                targetBox.Text = partialText
                task.wait(delayTime)
            end
        end
        if mySession ~= currentTypeSession then return end
        task.wait(0.02)
        
        targetBox:ReleaseFocus(true)
        
        task.spawn(function()
            pcall(function()
                local vim = game:GetService("VirtualInputManager")
                if vim then
                    vim:SendKeyEvent(true, Enum.KeyCode.Return, false, game)
                    task.wait(0.05)
                    vim:SendKeyEvent(false, Enum.KeyCode.Return, false, game)
                end
            end)
        end)

        task.delay(0.6, function()
            if not isMyTurn() and lastAttemptedWord ~= "" then
                dbHandler:autoLearn(lastAttemptedWord)
                if not MainDictMap[lastAttemptedWord] then
                    MainDictMap[lastAttemptedWord] = true
                    table.insert(WordDB, lastAttemptedWord)
                end
                lastAttemptedWord = ""
            end
        end)
    end)
end
