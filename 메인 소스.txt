-- 1. 의자(Seat) 파츠에 다른 플레이어나 인형이 앉아 있는지 판별하는 함수
local function isSeatOccupied(chaisePart)
    if not chaisePart then return true end
    if chaisePart:IsA("Seat") or chaisePart:IsA("VehicleSeat") then
        if chaisePart.Occupant then return true end
    end
    if chaisePart:IsA("BasePart") then
        for _, touchingPart in ipairs(chaisePart:GetTouchingParts()) do
            local character = touchingPart.Parent
            local humanoid = character and character:FindFirstChildOfClass("Humanoid")
            if humanoid then
                local player = Players:GetPlayerFromCharacter(character)
                if player and player ~= _lp then return true end
            end
        end
    end
    local chaisePos = chaisePart.Position
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= _lp and player.Character then
            local hrp = player.Character:FindFirstChild("HumanoidRootPart")
            if hrp and (hrp.Position - chaisePos).Magnitude < 2.5 then return true end
        end
    end
    return false
end

-- 2. 부드러운 텔레포트 (T_tp) 함수
local currentTween = nil
local function tweenTeleport(targetCFrame)
    local character = _lp.Character
    if not character then return end
    local rootPart = character:FindFirstChild("HumanoidRootPart")
    if not rootPart then return end

    local distance = (rootPart.Position - targetCFrame.Position).Magnitude
    local duration = math.clamp(distance / (config.tweenSpeed or 30), 0.1, 5)
    
    local tweenInfo = TweenInfo.new(duration, Enum.EasingStyle.Linear)
    if currentTween then currentTween:Cancel() end
    
    currentTween = TweenService:Create(rootPart, tweenInfo, {CFrame = targetCFrame})
    currentTween:Play()
end

-- 3. 게임 테이블 폴더 스캔 및 자동 이동(TP / Walk / T_tp) 실행 함수
local isTpCoolingDown = false 
local TP_COOLDOWN_TIME = 20.0

local function checkAndTeleportAllTables()
    if isTpCoolingDown then return end
    local baseFolder = Workspace:FindFirstChild("World") and Workspace.World:FindFirstChild("Scriptable") and Workspace.World.Scriptable:FindFirstChild("GameTable")
    if not baseFolder then return end
    local character = _lp.Character
    if not character then return end
    local rootPart = character:FindFirstChild("HumanoidRootPart")
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if not rootPart or not humanoid then return end

    local function scanFolder(parentFolder)
        for _, child in ipairs(parentFolder:GetChildren()) do
            local chairsFolder = child:FindFirstChild("Chairs")
            if chairsFolder then
                local emptySeats = {}
                local occupiedCount = 0
                for _, playerChairFolder in ipairs(chairsFolder:GetChildren()) do
                    local chaise = playerChairFolder:FindFirstChild("chaise")
                    if chaise then
                        if isSeatOccupied(chaise) then occupiedCount = occupiedCount + 1
                        else table.insert(emptySeats, chaise) end
                    end
                end
                local isFourPlayer = child.Name:match("4Player") or (#chairsFolder:GetChildren() >= 4)
                local triggerCondition = isFourPlayer and (occupiedCount >= 3) or (not isFourPlayer and occupiedCount >= 1)
                if triggerCondition and #emptySeats > 0 then
                    local alreadyNear = false
                    for _, playerChairFolder in ipairs(chairsFolder:GetChildren()) do
                        local chaise = playerChairFolder:FindFirstChild("chaise")
                        if chaise and (rootPart.Position - chaise.Position).Magnitude < 4.0 then alreadyNear = true break end
                    end
                    if not alreadyNear then
                        local targetSeat = emptySeats[1]
                        local targetCFrame = targetSeat.CFrame + Vector3.new(0, 3, 0)
                        
                        -- 설정 모드(tp, walk, T_tp)에 따른 자동앉기 이동 방식 처리
                        if config.autoSitMode == "walk" then
                            humanoid:MoveTo(targetSeat.Position)
                        elseif config.autoSitMode == "T_tp" then
                            tweenTeleport(targetCFrame)
                        else -- "tp" 기본값
                            rootPart.CFrame = targetCFrame
                        end

                        isTpCoolingDown = true
                        task.delay(TP_COOLDOWN_TIME, function() isTpCoolingDown = false end)
                        return true
                    end
                end
            else
                if #child:GetChildren() > 0 then
                    local result = scanFolder(child)
                    if result then return true end
                end
            end
        end
        return false
    end
    scanFolder(baseFolder)
end

RunService.Heartbeat:Connect(function()
    if not config.autoSitEnabled then return end
    pcall(function() checkAndTeleportAllTables() end)
end)

local SBASE = 0xAC00
local LCOUNT = 19
local VCOUNT = 21
local TCOUNT = 28
local NCOUNT = VCOUNT * TCOUNT

local function checkIsAttackWord(w)
    if not w or w == "" then return false end
    local lastChar = string.sub(w, -3)
    local code = utf8.codepoint(lastChar)
    if not code or code < SBASE or code > (SBASE + LCOUNT * NCOUNT - 1) then return false end
    local SIndex = code - SBASE
    local TIndex = SIndex % TCOUNT
    return TIndex > 0
end

local function getPriorityScore(w)
    if not w then return 0 end
    local score = 10
    if P_TIER1[w] or (type(w) == "string" and P_TIER1[string.sub(w, -3)]) then score = score + 1000
    elseif P_TIER2[w] or (type(w) == "string" and P_TIER2[string.sub(w, -3)]) then score = score + 500
    elseif P_TIER3[w] or (type(w) == "string" and P_TIER3[string.sub(w, -3)]) then score = score + 200 end
    
    if checkIsAttackWord(w) then score = score + 50 end
    score = score + (utf8.len(w) or #w) * 2
    return score
end

local function getBotSelectedWord(candidateList)
    if not candidateList or #candidateList == 0 then return nil end
    local bestWord = nil
    local highestScore = -1
    for _, word in ipairs(candidateList) do
        local currentScore = getPriorityScore(word)
        if currentScore > highestScore then
            highestScore = currentScore
            bestWord = word
        end
    end
    return bestWord
end

local dictionary = {}

local function findWord(startLetter)
    local candidates = {}
    if dictionary[startLetter] then
        for _, word in ipairs(dictionary[startLetter]) do
            if not usedWords[word] then table.insert(candidates, word) end
        end
    end
    if #candidates == 0 then return nil end
    local selected = getBotSelectedWord(candidates)
    if selected then usedWords[selected] = true return selected end
    return nil
end

local function applyDueumRule(char)
    local dueumMap = {
        ["라"] = "나", ["락"] = "낙", ["란"] = "난", ["람"] = "남", ["랍"] = "납", ["랑"] = "낭",
        ["래"] = "내", ["랭"] = "냉", ["렝"] = "넹", ["려"] = "여", ["녁"] = "역", ["년"] = "연",
        ["렬"] = "열", ["렴"] = "염", ["렵"] = "엽", ["령"] = "영", ["로"] = "노", ["록"] = "녹",
        ["론"] = "논", ["롱"] = "농", ["뢰"] = "뇌", ["루"] = "누", ["류"] = "유", ["륙"] = "육",
        ["륜"] = "윤", ["률"] = "율", ["륭"] = "융", ["리"] = "이", ["린"] = "인", ["림"] = "임"
    }
    return dueumMap[char] or char
end

local function getPriorityScoreEx(word, targetChar)
    if not word then return 0 end
    local score = 10
    if P_TIER1[word] or (type(word) == "string" and P_TIER1[string.sub(word, -3)]) then score = score + 1000
    elseif P_TIER2[word] or (type(word) == "string" and P_TIER2[string.sub(word, -3)]) then score = score + 500
    elseif P_TIER3[word] or (type(word) == "string" and P_TIER3[string.sub(word, -3)]) then score = score + 200 end
    
    if checkIsAttackWord(word) then score = score + 50 end
    local length = utf8.len(word) or #word
    score = score + length * 2
    if isExcludedEndingWord(string.sub(word, -3)) then score = score - 100 end
    return score
end

local function getBotSelectedWordEx(targetChar, dictionary)
    local bestWord = nil
    local maxScore = -9999
    local searchChar = applyDueumRule(targetChar)
    if dictionary[searchChar] then
        for _, word in ipairs(dictionary[searchChar]) do
            local score = getPriorityScoreEx(word, searchChar)
            if score > maxScore then
                maxScore = score
                bestWord = word
            end
        end
    end
    return bestWord
end

        local ch = string.match(text, "'(.-)'") or string.match(text, '"(.-)"')
        if ch and ch ~= "" then return ch end
        local hangulMatches = {}
        for match in string.gmatch(text, "[\224-\239][\128-\1bf][\128-\1bf]") do
            table.insert(hangulMatches, match)
        end
        if #hangulMatches > 0 then return hangulMatches[#hangulMatches] end
    end
    return nil
end

local TypewriterHandler = {}

function TypewriterHandler.new(fileName)
    local self = setmetatable({}, TypewriterHandler)
    self.fileName = fileName or "WordDB.json"
    self.learnedWords = self:loadDatabase()
    return self
end

function TypewriterHandler:loadDatabase()
    if isfile and isfile(self.fileName) then
        local success, data = pcall(function() return HttpService:JSONDecode(readfile(self.fileName)) end)
        if success and type(data) == "table" then return data end
    end
    return {}
end

function TypewriterHandler:saveDatabase()
    if writefile then
        local encoded = HttpService:JSONEncode(self.learnedWords)
        writefile(self.fileName, encoded)
    end
end

function TypewriterHandler:autoLearn(rawWord)
    if not config.learningEnabled then return end
    if type(rawWord) ~= "string" then return end
    local word = string.lower(string.gsub(rawWord, "^%s*(.-)%s*$", "%1"))
    if word == "" then return end
    if not self.learnedWords[word] then
        self.learnedWords[word] = true
        self:saveDatabase()
        return true
    end
    return false
end

local function parseWordsFromUrlData(rawText)
    if not rawText or rawText == "" then return end
    local success, decoded = pcall(function() return HttpService:JSONDecode(rawText) end)
    if success and type(decoded) == "table" then
        for _, v in pairs(decoded) do
            if type(v) == "string" and v ~= "" then
                if not MainDictMap[v] then MainDictMap[v] = true table.insert(WordDB, v) end
            elseif type(v) == "table" then
                for _, subV in pairs(v) do
                    if type(subV) == "string" and subV ~= "" then
                        if not MainDictMap[subV] then MainDictMap[subV] = true table.insert(WordDB, subV) end
                    end
                end
            end
        end
        return
    end

    for word in rawText:gmatch('"([^"]+)"') do
        local cleaned = word:match("^%s*(.-)%s*$")
        if cleaned and cleaned ~= "" and not cleaned:find("http") and not cleaned:find(":") then
            if not MainDictMap[cleaned] then MainDictMap[cleaned] = true table.insert(WordDB, cleaned) end
        end
    end

    for line in rawText:gmatch("[^\r\n]+") do
        local cleaned = line:match("^%s*(.-)%s*$")
        if cleaned and cleaned ~= "" and not MainDictMap[cleaned] and not cleaned:find("http") then
            MainDictMap[cleaned] = true
            table.insert(WordDB, cleaned)
        end
    end
end

JONG_KEYS[9]={0x3139,0x3131}; JONG_KEYS[10]={0x3139,0x3141}; JONG_KEYS[11]={0x3139,0x3142}
JONG_KEYS[12]={0x3139,0x3145}; JONG_KEYS[13]={0x3139,0x314C}; JONG_KEYS[14]={0x3139,0x314D}
JONG_KEYS[15]={0x3139,0x314E}; JONG_KEYS[18]={0x3142,0x3145}

local CHO_OF, JUNG_OF, JONG_OF = {}, {}, {}
for i, c in ipairs(CHO) do CHO_OF[c] = i - 1 end
for i, c in ipairs(JUNG) do JUNG_OF[c] = i - 1 end
for i = 1, 27 do if #JONG_KEYS[i] == 1 then JONG_OF[JONG_KEYS[i][1]] = i end end

local JONG_COMBINE, JUNG_COMBINE = {}, {}
for i = 1, 27 do local k = JONG_KEYS[i] if #k == 2 then JONG_COMBINE[JONG_OF[k[1]]*100 + k[2]] = i end end
for i = 0, 20 do local k = JUNG_KEYS[i] if #k == 2 then JUNG_COMBINE[JUNG_OF[k[1]]*1000 + k[2]] = i end end

local function wordToKeys(word)
    local keys = {}
    for _, cp in utf8.codes(word) do
        if cp >= SBASE_CONST and cp < SBASE_CONST + SCOUNT_CONST then
            local s = cp - SBASE_CONST
            keys[#keys+1] = CHO[math.floor(s / NCOUNT_CONST) + 1]
            for _, k in ipairs(JUNG_KEYS[math.floor((s % NCOUNT_CONST) / TCOUNT_CONST)]) do keys[#keys+1] = k end
            local t = s % TCOUNT_CONST
            if t > 0 then for _, k in ipairs(JONG_KEYS[t]) do keys[#keys+1] = k end end
        else keys[#keys+1] = cp end
    end
    return keys
end

local function render(committed, cho, jung, jong)
    if cho and jung then return committed .. utf8.char(SBASE_CONST + (cho*VCOUNT_CONST + jung)*TCOUNT_CONST + jong)
    elseif cho then return committed .. utf8.char(CHO[cho+1])
    elseif jung then return committed .. utf8.char(JUNG[jung+1]) end
    return committed
end

local function buildTypeStates(word)
    local keys = wordToKeys(word)
    local states, committed = {}, ""
    local cho, jung, jong = nil, nil, 0
    local function commit() committed = render(committed, cho, jung, jong); cho, jung, jong = nil, nil, 0 end
    for _, cp in ipairs(keys) do
        if CHO_OF[cp] then
            if cho and jung then
                if jong == 0 and JONG_OF[cp] then jong = JONG_OF[cp]
                elseif jong > 0 and JONG_COMBINE[jong*100 + cp] then jong = JONG_COMBINE[jong*100 + cp]
                else commit(); cho = CHO_OF[cp] end
            elseif cho then commit(); cho = CHO_OF[cp]
            else cho = CHO_OF[cp] end
        elseif JUNG_OF[cp] then
            if jong > 0 then
                local kt = JONG_KEYS[jong]
                if #kt == 2 then jong = JONG_OF[kt[1]]; commit(); cho = CHO_OF[kt[2]]; jung = JUNG_OF[cp]
                else jong = 0; commit(); cho = CHO_OF[kt[1]]; jung = JUNG_OF[cp] end
            elseif cho and jung then
                local comb = JUNG_COMBINE[jung*1000 + cp]
                if comb then jung = comb else commit(); jung = JUNG_OF[cp] end
            elseif jung then
                local comb = JUNG_COMBINE[jung*1000 + cp]
                if comb then jung = comb else commit(); jung = JUNG_OF[cp] end
            else jung = JUNG_OF[cp] end
        else commit(); committed = committed .. utf8.char(cp) end
        states[#states+1] = render(committed, cho, jung, jong)
    end
    local out = {}
    for _, s in ipairs(states) do if out[#out] ~= s then out[#out+1] = s end end
    if #out == 0 or out[#out] ~= word then out[#out+1] = word end
    return out
end

local function analyzeWordPrediction(startWord, depth)
    depth = depth or 1
    local lastChar = ""
    for _, cp in utf8.codes(startWord) do lastChar = utf8.char(cp) end
    local candidates = {}
    for _, w in ipairs(WordDB) do
        if utf8.len(w) >= 1 and utf8.char(utf8.codepoint(w, 1, 1)) == lastChar then
            table.insert(candidates, w)
        end
    end
    if #candidates == 0 then return "지는 수", depth end
    for _, w in ipairs(candidates) do
        local len = utf8.len(w)
        local isAtk = checkIsAttackWord(w)
        local isSuper = (len >= 6)
        if isAtk and isSuper then return "한방 초장문", depth
        elseif isAtk then return "한방", depth
        elseif isSuper then return "초장문", depth end
        if depth < config.readingDepth then
            local res, d = analyzeWordPrediction(w, depth + 1)
            if res == "한방" or res == "한방 초장문" then return "반격", d end
        end
    end
    return "완벽방어", depth
end

local function sendChatMessage(msg)
    if msg == "" then return end
    if (UserRank == "Premium" or UserRank == "Admin") and config.antiBanEnabled then
        task.spawn(function()
            task.wait(1.0)
            pcall(function()
                local chatEvents = ReplicatedStorage:FindFirstChild("DefaultChatSystemChatEvents")
                local removeMsg = chatEvents and chatEvents:FindFirstChild("RemoveMessageRequest")
                if removeMsg then removeMsg:FireServer(msg) end
            end)
        end)
    end
    if TextChatService.ChatVersion == Enum.ChatVersion.TextChatService then
        local channel = TextChatService:FindFirstChild("TextChannels") and TextChatService.TextChannels:FindFirstChild("RBXGeneral")
        if channel then channel:SendAsync(msg) end
    else
        local chatEvents = ReplicatedStorage:FindFirstChild("DefaultChatSystemChatEvents")
        local sendMsg = chatEvents and chatEvents:FindFirstChild("SayMessageRequest")
        if sendMsg then sendMsg:FireServer(msg, "All") end
    end
end

local lastAttemptedWord = ""
local function sendWordToGame(word)
    if word == "" then return end
    local targetBox = forceLoadGameTextBox()
    if not targetBox or not targetBox.Visible then return end
    
    lastAttemptedWord = word
    currentTypeSession = currentTypeSession + 1
    local mySession = currentTypeSession
    local targetRealSpeed = tonumber(config.typingSpeed) or 350

    pcall(function()
        targetBox.Visible = true
        if targetBox.Parent and targetBox.Parent:IsA("GuiObject") then targetBox.Parent.Visible = true end
        targetBox:CaptureFocus()

        if targetRealSpeed <= 0 then
            if mySession ~= currentTypeSession then return end
            targetBox.Text = word
        else
            local totalStates = buildTypeStates(word)
            local delayTime = 60 / (targetRealSpeed * 2.8) 
            for _, partialText in ipairs(totalStates) do
                if mySession ~= currentTypeSession then return end
                targetBox.Text = partialText
                task.wait(delayTime)
            end
        end
        if mySession ~= currentTypeSession then return end
        task.wait(0.02)
        targetBox:ReleaseFocus(true)
        
        task.spawn(function()
            pcall(function()
                local vim = game:GetService("VirtualInputManager")
                if vim then
                    vim:SendKeyEvent(true, Enum.KeyCode.Return, false, game)
                    task.wait(0.05)
                    vim:SendKeyEvent(false, Enum.KeyCode.Return, false, game)
                end
            end)
        end)

        task.delay(0.6, function()
            if not isMyTurn() and lastAttemptedWord ~= "" then
                dbHandler:autoLearn(lastAttemptedWord)
                if not MainDictMap[lastAttemptedWord] then
                    MainDictMap[lastAttemptedWord] = true
                    table.insert(WordDB, lastAttemptedWord)
                end
                lastAttemptedWord = ""
            end
        end)
    end)
end
